\section{Variationally--consistent reproducing kernel gradient smoothing meshfree formulation}\label{sec:rkgs}

Section 3 states that the variational consistency plays a key role for the accuracy of Galerkin meshfree methods.
The conventional Gauss integration, however, can only fulfil the variational consistency of order $q=1$ and is therefore not able to verify the error estimates presented in Section \ref{sec:error}.
In this section the reproducing kernel gradient smoothing integration (RKGSI) \cite{} based on the Hellinger--Reissner variational principle \cite{} is adopted as the variationally consistent integration scheme, in which the bending moment $\boldsymbol M$ is treated as an independent field.
The corresponding boundary value problem reads
\begin{equation}\label{hr_strong}
	\begin{cases}
		M_{ij,ij}+\bar q=0 & \text{in}\;\Omega\\
		V(\boldsymbol M)=M_{ni,i}+M_{ns,s}=\bar V & \text{on}\;\Gamma_{V}\\
		M(\boldsymbol M)=M_{nn}=\bar M & \text{on}\;\Gamma_{M}\\
		P(\boldsymbol M)=-\left[M_{ns}\right]=\bar P & \text{at}\;c_P\\
		w=\bar w & \text{on}\;\Gamma_{w}\\
		\theta(w)=w_{,n}=\bar\theta & \text{on}\;\Gamma_{\theta}\\
		w=\bar w & \text{at}\;c_w
	\end{cases}
\end{equation}

The Hellinger--Reissner functional of the Kirchhoff plate then reads
\begin{equation}\label{hr_functional}
	\begin{aligned}
		\Pi(\boldsymbol M,w)={}&\int_\Omega \frac{1}{2} M_{ij} C^{-1}_{ijkl} M_{kl} d\Omega
		-\int_{\Gamma_{w}} V(\boldsymbol M) \bar w d\Gamma
		+\int_{\Gamma_{\theta}} M(\boldsymbol M) \bar\theta d\Gamma
		-P(\boldsymbol M)\bar w\vert_{\boldsymbol x\in c_w} \\
		&+\int_\Omega w\left(M_{ij,ij}+\bar q\right)d\Omega
		+\int_{\Gamma_{M}} \theta(w)\left(M(\boldsymbol M)-\bar M\right)d\Gamma \\
		&-\int_{\Gamma_{V}} w\left(V(\boldsymbol M)-\bar V\right)d\Gamma
		-w\left(P(\boldsymbol M)-\bar P\right)\vert_{\boldsymbol x\in c_P}
	\end{aligned}
\end{equation}
where $C^{-1}_{ijkl}$ is the inverse of the elasticity tensor $C_{ijkl}$ given in Eq. \eqref{curvature}, i.e.
\begin{equation}\label{compliance}
	C_{ijkl}C^{-1}_{klmn}=\frac{1}{2}\left(\delta_{im}\delta_{jn}+\delta_{in}\delta_{jm}\right)
\end{equation}
Taking the variation of Eq. \eqref{hr_functional} with respect to both $\boldsymbol M$ and $w$, and requiring $\delta\Pi=0$ for arbitrary $\delta\boldsymbol M$ and $\delta w$, yields
\begin{equation}\label{hr_variation_M}
	\begin{aligned}
		&\int_\Omega \delta M_{ij} C^{-1}_{ijkl} M_{kl} d\Omega
		+\int_\Omega w\,\delta M_{ij,ij}d\Omega
		-\int_{\Gamma_{w}}V(\delta\boldsymbol M)\bar w d\Gamma
		+\int_{\Gamma_{\theta}}M(\delta\boldsymbol M)\bar\theta d\Gamma
		-P(\delta\boldsymbol M)\bar w\vert_{\boldsymbol x\in c_w} \\
		&\qquad+\int_{\Gamma_{M}}\theta(w)M(\delta\boldsymbol M)d\Gamma
		-\int_{\Gamma_{V}}w V(\delta\boldsymbol M)d\Gamma
		-w P(\delta\boldsymbol M)\vert_{\boldsymbol x\in c_P}=0
	\end{aligned}
\end{equation}
\begin{equation}\label{hr_variation_w}
	\begin{aligned}
		&\int_\Omega \delta w\left(M_{ij,ij}+\bar q\right)d\Omega
		+\int_{\Gamma_{M}}\theta(\delta w)\left(M(\boldsymbol M)-\bar M\right)d\Gamma
		-\int_{\Gamma_{V}}\delta w\left(V(\boldsymbol M)-\bar V\right)d\Gamma \\
		&\qquad-\delta w\left(P(\boldsymbol M)-\bar P\right)\vert_{\boldsymbol x\in c_P}=0
	\end{aligned}
\end{equation}
It is noted that the natural boundary conditions in Eq. \eqref{hr_strong} are now enforced in the sense of the Lagrange multiplier method, with the deflection $w$ acting as the multiplier.
On the other hand, the essential boundary conditions $w=\bar w$ and $\theta(w)=\bar\theta$ enter Eq. \eqref{hr_functional} as the natural conditions of the mixed formulation, so that they are accommodated without introducing any stabilization parameter.

In the context of the HR-based RKGSI framework, the problem domain is partitioned into a set of integration cells $\Omega_C$, i.e. $\Omega=\bigcup_{C=1}^{n_e}\Omega_C$, where $n_e$ denotes the total number of integration cells.
The bending moment is approximated in each integration cell $\Omega_C$ by a polynomial of degree $q-2$
\begin{equation}\label{moment_approx}
	M^h_{ij}(\boldsymbol x)=\boldsymbol a_{ij}^{T}\boldsymbol p^{[q-2]}(\boldsymbol x)
\end{equation}
where the constant coefficient vector $\boldsymbol a_{ij}$ is determined by Eq. \eqref{hr_variation_M}:
\begin{equation}\label{moment_coef}
	\boldsymbol a_{ij}=-C_{ijkl}\boldsymbol G^{-1}\left(\sum_{I=1}^{n_p}\left(\tilde{\boldsymbol g}_{klI}-\bar{\boldsymbol g}_{klI}\right)d_I+\hat{\boldsymbol g}_{kl}\right),\qquad
	\boldsymbol G=\int_{\Omega_C}\boldsymbol p^{[q-2]}\boldsymbol p^{[q-2]T}d\Omega
\end{equation}
with
\begin{equation}\label{g_ij}
	\left\{
	\begin{aligned}
		\tilde{\boldsymbol g}_{ijI}&=\int_{\Gamma_C}\Psi_{I,n}\boldsymbol p^{[q-2]} n_i n_j d\Gamma
		-\int_{\Gamma_C}\Psi_I\left(n_i\boldsymbol p^{[q-2]}_{,j}+\boldsymbol p^{[q-2]}_{,k}s_i n_j s_k\right)d\Gamma
		+\left[\Psi_I\boldsymbol p^{[q-2]} n_i s_j\right]_{\boldsymbol x\in c_C}
		+\int_{\Omega_C}\Psi_I\boldsymbol p^{[q-2]}_{,ij}d\Omega \\
		\bar{\boldsymbol g}_{ijI}&=\int_{\Gamma_C\cap\Gamma_{\theta}}\Psi_{I,n}\boldsymbol p^{[q-2]} n_i n_j d\Gamma
		-\int_{\Gamma_C\cap\Gamma_{w}}\Psi_I\left(n_i\boldsymbol p^{[q-2]}_{,j}+\boldsymbol p^{[q-2]}_{,k}s_i n_j s_k\right)d\Gamma
		+\left[\Psi_I\boldsymbol p^{[q-2]} n_i s_j\right]_{\boldsymbol x\in c_C\cap c_w} \\
		\hat{\boldsymbol g}_{ij}&=\int_{\Gamma_C\cap\Gamma_{\theta}}\boldsymbol p^{[q-2]} n_i n_j \bar\theta d\Gamma
		-\int_{\Gamma_C\cap\Gamma_{w}}\left(n_i\boldsymbol p^{[q-2]}_{,j}+\boldsymbol p^{[q-2]}_{,k}s_i n_j s_k\right)\bar w d\Gamma
		+\bar w\left[\boldsymbol p^{[q-2]} n_i s_j\right]_{\boldsymbol x\in c_C\cap c_w}
	\end{aligned}
	\right.
\end{equation}
Substituting Eq. \eqref{moment_coef} into Eq. \eqref{moment_approx} gives
\begin{equation}\label{moment_discrete}
	\boldsymbol M^h = \tilde{\boldsymbol M}^h - \bar{\boldsymbol M}^h + \hat{\boldsymbol M}^h = \boldsymbol C:(\tilde{\boldsymbol \kappa}^h - \bar{\boldsymbol \kappa}^h + \hat{\boldsymbol \kappa}^h)
\end{equation}
where
\begin{equation}\label{kappa_decomposition}
	\begin{aligned}
		\tilde\kappa^h_{ij}&=-\sum_{I=1}^{n_p}\tilde\Psi_{I,ij} d_I,&\qquad
		\bar\kappa^h_{ij}&=-\sum_{I=1}^{n_p}\bar \Psi_{I,ij} d_I,&\qquad
		\hat\kappa^h_{ij}&=-\boldsymbol p^{[q-2]T}\boldsymbol G^{-1}\hat{\boldsymbol g}_{ij}
	\end{aligned}
\end{equation}
with
\begin{equation}\label{smoothed_gradient}
	\tilde\Psi_{I,ij}(\boldsymbol x)=\boldsymbol p^{[q-2]T}\boldsymbol G^{-1}\tilde{\boldsymbol g}_{ijI},\qquad
	\bar \Psi_{I,ij}(\boldsymbol x)=\boldsymbol p^{[q-2]T}\boldsymbol G^{-1}\bar{\boldsymbol g}_{ijI}
\end{equation}
With the curvature components in Eq. \eqref{kappa_decomposition}, the bending moment is finally expressed in terms of the nodal coefficients $d_I$ of the deflection.
The smoothed gradients satisfy the following consistency condition \cite{} within each integration cell $\Omega_C$
\begin{equation}\label{consistency_hr}:
	\begin{aligned}
		\sum_{I=1}^{n_p}\tilde\Psi_{I,ij}(\boldsymbol x)\boldsymbol p^{[q-2]}(\boldsymbol x_I)=\boldsymbol p^{[q-2]}_{,ij}(\boldsymbol x)
	\end{aligned}
\end{equation}
Substituting Eqs. \eqref{moment_discrete} and \eqref{kappa_decomposition} into the weak form \eqref{hr_variation_w} then leads to the following variational problem of the RKGSI.

\begin{equation}\label{variation_rk}
	\text{Find}\; w^h \in W^h:\quad\tilde a(\delta w^h, w^h) = \tilde f(\delta w^h) ,\quad \forall \delta w \in W^h
\end{equation}
\begin{equation}\label{variation_rk_a}
	\begin{aligned}
		\tilde a(\delta w^h,\delta w^h) &= \int_\Omega \delta\tilde{\boldsymbol\kappa}^h : \boldsymbol C : \tilde{\boldsymbol\kappa}^h d\Omega
		- \int_{\Gamma_{w}} \left( V(\delta\bar{\boldsymbol M}^h) w^h + V(\bar{\boldsymbol M}^h) \delta w^h \right) d\Gamma \\
		&+ \int_{\Gamma_{\theta}} \left( M(\delta\bar{\boldsymbol M}^h) \theta(w^h) + M(\bar{\boldsymbol M}^h) \theta(\delta w^h) \right) d\Gamma 
		- \left( P(\delta\bar{\boldsymbol M}^h) w^h + P(\bar{\boldsymbol M}^h) \delta w^h \right) \vert_{\boldsymbol x \in c_w} \\
		&+ \int_{\Gamma_{w}} V(\delta\hat{\boldsymbol M}^h) w^h d\Gamma 
		- \int_{\Gamma_{\theta}} M(\delta\hat{\boldsymbol M}^h) \theta(w^h) d\Gamma 
		+ P(\delta\hat{\boldsymbol M}^h) w^h \vert_{\boldsymbol x \in c_w} \\
	\end{aligned}
\end{equation}
with
\begin{equation}\label{variation_rk_f}
	\begin{aligned}
		\tilde f(\delta w) &= f(\delta w^h)  
		- \int_{\Gamma_{w}} V(\delta\tilde{\boldsymbol M}^h) \bar w d\Gamma 
		+ \int_{\Gamma_{\theta}} M(\delta\tilde{\boldsymbol M}^h) \bar \theta d\Gamma 
		- P(\delta\tilde{\boldsymbol M}^h) \bar w \vert_{\boldsymbol x \in c_w} \\
		&+ \int_{\Gamma_{w}} V(\delta\hat{\boldsymbol M}^h) \bar w d\Gamma 
		- \int_{\Gamma_{\theta}} M(\delta\hat{\boldsymbol M}^h) \bar \theta d\Gamma 
		+ P(\delta\hat{\boldsymbol M}^h) \bar w \vert_{\boldsymbol x \in c_w} \\
	\end{aligned}
\end{equation}

It can be found from Eqs. \eqref{variation_rk}--\eqref{variation_rk_f} that the rearranged variational problem of the RKGSI takes a similar form to the Nitsche-based Galerkin meshfree formulation \cite{} in Eq. \eqref{nitsche_weakform}, in the sense that both consist of a consistent term and a stabilized term.
Compared with the Nitsche method, the consistent term recovers the $q$-th order variational consistency owing to the cell-wise polynomial approximation of the bending moment, and the resulting orthogonality estimate is given by Eq. \eqref{orthogonality_estimate}.
On the other hand, the stabilized term no longer relies on the penalty method, so that the coercivity of $\tilde a$ is guaranteed without any artificial parameter.
